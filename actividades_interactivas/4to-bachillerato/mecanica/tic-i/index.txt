    <script>
        // Tab switching logic
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('tab-active');
                btn.classList.add('text-slate-400');
            });

            document.getElementById(tabId).classList.remove('hidden');
            const activeBtn = document.getElementById('nav-' + tabId);
            if (activeBtn) {
                activeBtn.classList.add('tab-active');
                activeBtn.classList.remove('text-slate-400');
            }

            if (tabId === 'tab-entregable') {
                updatePDFPreview();
            }
        }

        // Modal Helpers
        function openModal(modalId) {
            document.getElementById(modalId).classList.remove('hidden');
        }
        function closeModal(modalId) {
            document.getElementById(modalId).classList.add('hidden');
        }

        // Toast Helper
        function showToast(text, iconClass = "fa-solid fa-circle-check text-emerald-400") {
            const toast = document.getElementById('toast-message');
            document.getElementById('toast-text').innerText = text;
            document.getElementById('toast-icon').className = iconClass + " text-base";
            toast.classList.remove('hidden');
            setTimeout(() => toast.classList.add('hidden'), 3500);
        }

        // Word count helper
        function countWords(str) {
            return str.trim() ? str.trim().split(/\s+/).length : 0;
        }

        // --- VALIDATION ENGINE FOR ACTIVITIES --- //

        // Actividad 1 Validation
        function checkAct1Validation() {
            const title = document.getElementById('form-title-input').value;
            const text = document.getElementById('act1-text').value;
            const wordCount = countWords(text);
            
            document.getElementById('act1-word-count').innerText = `Palabras: ${wordCount}`;

            const hasHeading = text.toLowerCase().includes('actividad 1:');
            const isLongEnough = wordCount >= 30;

            let score = 0;
            if (title.length > 5) score++;
            if (hasHeading) score++;
            if (wordCount >= 20) score++;
            if (wordCount >= 40) score++;
            if (text.toLowerCase().includes('repuesto') || text.toLowerCase().includes('taller') || text.toLowerCase().includes('error')) score++;

            document.getElementById('act1-count').innerText = Math.min(score, 5);

            const statusEl = document.getElementById('act1-status');
            const kwCheck = document.getElementById('act1-kw-check');

            if (hasHeading && isLongEnough) {
                statusEl.innerText = "Completado (4/4 pts)";
                statusEl.className = "font-bold text-emerald-400";
                kwCheck.innerHTML = '<i class="fa-solid fa-circle-check text-emerald-400"></i> Encabezado y análisis adecuados';
                kwCheck.className = "text-emerald-400";
            } else {
                statusEl.innerText = "Incompleto";
                statusEl.className = "font-bold text-amber-400";
                kwCheck.innerHTML = '<i class="fa-solid fa-circle-xmark text-rose-400"></i> Falta título "Actividad 1:" o análisis (mínimo 30 palabras)';
                kwCheck.className = "text-rose-400";
            }

            calculateGlobalScore();
        }

        // Actividad 2 Validation
        function checkAct2Validation() {
            const text = document.getElementById('act2-text').value;
            const wordCount = countWords(text);
            
            document.getElementById('act2-word-count').innerText = `Palabras: ${wordCount}`;

            const hasHeading = text.toLowerCase().includes('actividad 2:');

            document.getElementById('act2-count').innerText = Math.min(score, 5);

        }

                kwCheck.className = "text-rose-400";
            }

            calculateGlobalScore();

        // Calculate total rubric points (10 pts total)
        function calculateGlobalScore() {
            const t1 = document.getElementById('act1-text').value;
            const t2 = document.getElementById('act2-text').value;
            const t3 = document.getElementById('act3-text').value;

            let points = 0;

            // Act 1 (4 pts max)
            if (t1.toLowerCase().includes('actividad 1:') && countWords(t1) >= 30) {
                points += 4;
            } else if (t1.trim().length > 10) {
                points += 2;
            }

            // Act 2 (3 pts max)
            if (t2.toLowerCase().includes('actividad 2:') && countWords(t2) >= 25) {
                points += 3;
            } else if (t2.trim().length > 10) {
                points += 1;
            }

            // Act 3 (3 pts max)
            if (t3.toLowerCase().includes('actividad 3:') && countWords(t3) >= 25) {
                points += 3;
            } else if (t3.trim().length > 10) {
                points += 1;
            }

            // Update Header progress
            document.getElementById('global-progress-bar').style.width = (points * 10) + "%";
            document.getElementById('global-progress-text').innerText = `${points}/10 pts`;

            // Update Badge
            const badge = document.getElementById('score-badge');
            const scoreDesc = document.getElementById('score-description');
            badge.innerText = `${points}/10`;

            if (points === 10) {
                badge.className = "w-14 h-14 rounded-full bg-emerald-500/20 border-2 border-emerald-500 text-emerald-400 font-extrabold text-lg flex items-center justify-center";
                scoreDesc.innerText = "¡Excelente! Has cumplido al 100% todos los criterios de la Rúbrica Oficial.";
            } else if (points >= 6) {
                badge.className = "w-14 h-14 rounded-full bg-amber-500/20 border-2 border-amber-500 text-amber-400 font-extrabold text-lg flex items-center justify-center";
                scoreDesc.innerText = "Nivel Aceptable. Revisa las sugerencias de redacción para alcanzar los 10 puntos.";
            } else {
                badge.className = "w-14 h-14 rounded-full bg-rose-500/20 border-2 border-rose-500 text-rose-400 font-extrabold text-lg flex items-center justify-center";
                scoreDesc.innerText = "Puntaje insuficiente. Asegúrate de completar el texto escrito exigido en cada actividad.";
            }
        }

        // Live Document Preview Update
        function updatePDFPreview() {
            const name = document.getElementById('student-name').value || "[Escribe tu Nombre Completo]";
            const key = document.getElementById('student-key').value || "[Clave y Sección]";

            document.getElementById('pdf-preview-name').innerText = name;
            document.getElementById('pdf-preview-key').innerText = key;

            const t1 = document.getElementById('act1-text').value || "Pendiente de redacción en Pestaña 2 (Actividad 1)...";
            const t2 = document.getElementById('act2-text').value || "Pendiente de redacción en Pestaña 3 (Actividad 2)...";
            const t3 = document.getElementById('act3-text').value || "Pendiente de redacción en Pestaña 4 (Actividad 3)...";

            document.getElementById('pdf-act1-body').innerText = t1;
            document.getElementById('pdf-act2-body').innerText = t2;
            document.getElementById('pdf-act3-body').innerText = t3;
        }

        // Download Final PDF function using html2pdf
        function downloadAsPDF() {
            const studentName = document.getElementById('student-name').value.trim();
            if (!studentName) {
                showToast("Por favor escribe tu nombre completo antes de generar el PDF", "fa-solid fa-triangle-exclamation text-amber-400");
                switchTab('tab-entregable');
                document.getElementById('student-name').focus();
                return;
            }

            showToast("Generando archivo PDF técnico...", "fa-solid fa-gear fa-spin text-amber-400");

            const element = document.getElementById('pdf-document-area');
            const opt = {
                margin:       0.5,
                filename:     `Guia5_TIC_I_${studentName.replace(/\s+/g, '_')}.pdf`,
                image:        { type: 'jpeg', quality: 0.98 },
                html2canvas:  { scale: 2 },
                jsPDF:        { unit: 'in', format: 'letter', orientation: 'portrait' }
            };

            html2pdf().set(opt).from(element).save().then(() => {
                showToast("PDF descargado exitosamente. ¡Listo para Google Classroom!", "fa-solid fa-circle-check text-emerald-400");
            }).catch(err => {
                showToast("Error al exportar. Puedes presionar Ctrl+P para guardar como PDF.", "fa-solid fa-circle-exclamation text-rose-400");
            });
        }
